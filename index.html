export default async function handler(req, res) { res.setHeader( "Access-Control-Allow-Origin", "*" ); res.setHeader( "Access-Control-Allow-Methods", "POST, OPTIONS" ); res.setHeader( "Access-Control-Allow-Headers", "Content-Type" ); if (req.method === "OPTIONS") { return res.status(200).end(); } if (req.method !== "POST") { return res.status(405).json({ error: "Sadece POST isteği kullanılabilir." }); } try { const { image, category } = req.body || {}; if (!image) { return res.status(400).json({ error: "Fotoğraf gönderilmedi." }); } if (!category) { return res.status(400).json({ error: "Eşya türü seçilmedi." }); } if (!process.env.OPENAI_API_KEY) { return res.status(500).json({ error: "OpenAI anahtarı sunucuda tanımlı değil." }); } const prompt = ` Bu fotoğraftaki eşyayı dikkatlice incele. Eşya kategorisi: ${category} Türkiye'deki kullanıcı için Türkçe cevap ver. Şunları sırayla yaz: 1. Eşyanın ne olduğunu tahmin et. 2. Görülebiliyorsa marka ve modelini belirt. 3. Fotoğraftan görülebilen durumunu değerlendir. 4. Sıfır ürünse tahmini değer aralığını TL olarak yaz. 5. İkinci el ise tahmini satış fiyatı aralığını TL olarak yaz. 6. Koleksiyonluk veya antika olma ihtimali varsa belirt. 7. Fiyatı etkileyebilecek önemli ayrıntıları söyle. 8. Emin olmadığın bilgileri kesinmiş gibi yazma. Cevabı sade, anlaşılır ve çok uzun olmayacak şekilde ver. En sonda: "Tahmini Değer: ... TL - ... TL" şeklinde kısa bir özet yaz. Bu bir görsel tahmindir; kesin ekspertiz sonucu değildir. `; const openaiCevap = await fetch( "https://api.openai.com/v1/responses", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${process.env.OPENAI_API_KEY}` }, body: JSON.stringify({ model: "gpt-5.6-luna", input: [ { role: "user", content: [ { type: "input_text", text: prompt }, { type: "input_image", image_url: image } ] } ], max_output_tokens: 700 }) } ); const data = await openaiCevap.json(); if (!openaiCevap.ok) { console.error( "OpenAI hatası:", data ); return res.status(openaiCevap.status).json({ error: data?.error?.message || "Yapay zekâ servisine bağlanılamadı." }); } let metin = ""; if (Array.isArray(data.output)) { for (const item of data.output) { if ( item.type === "message" && Array.isArray(item.content) ) { for (const content of item.content) { if ( content.type === "output_text" && content.text ) { metin += content.text; } } } } } if (!metin) { return res.status(500).json({ error: "Yapay zekâ cevap verdi ancak sonuç okunamadı." }); } return res.status(200).json({ message: metin.trim() }); } catch (hata) { console.error(hata); return res.status(500).json({ error: "Sunucuda beklenmeyen bir hata oluştu." }); } }
